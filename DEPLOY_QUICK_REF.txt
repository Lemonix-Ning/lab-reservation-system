🚀 快速部署命令

# 1. SSH 连接

📝 手动部署命令（如果自动脚本失败）
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 安装 Docker
curl -fsSL https://get.docker.com | sh
systemctl enable docker && systemctl start docker

# 配置环境
cp .env.example .env
nano .env  # 填写上面的密钥

# 配置邮箱
nano init-ssl.sh  # 修改 EMAIL=""

# 执行部署
chmod +x init-ssl.sh update.sh
./init-ssl.sh

✅ 验证命令
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 检查容器
docker compose ps

# 查看日志
docker compose logs -f app

# 测试访问
curl -I https://lemonix.loc.cc

🔧 配置管理员
━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━━
# 1. 访问网站并 GitHub 登录
https://lemonix.loc.cc

# 2. 获取 openId
docker compose logs app | grep "openId\|github-"

# 3. 配置管理员
nano .env  # 填写 OWNER_OPEN_ID=github-xxxxxx
docker compose restart app
